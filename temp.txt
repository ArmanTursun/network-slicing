    def _calculate_safety_value(self, info_dict: Dict) -> float:
        """
        Calculates the safety value m(s') for the state outcomes described by info_dict.
        A higher value means safer. Safety is based on meeting GBRs and keeping queues manageable.
        The safety threshold 'h' (e.g., 0.0 or -0.5) would be applied to this value by SPPO.

        Args:
            info_dict (Dict): Dictionary containing outcome metrics for the new state s'.
                              Expected keys:
                              'achieved_th_per_slice': List/np.array of throughputs (Mbps)
                              'queue_len_per_slice': List/np.array of queue lengths (KB)
        Returns:
            float: The calculated safety value m(s').
        """
        achieved_th = np.array(info_dict.get('achieved_th_per_slice', np.zeros(self.num_slices)), dtype=np.float32)
        queue_len = np.array(info_dict.get('queue_len_per_slice', np.zeros(self.num_slices)), dtype=np.float32)

        # 1. GBR Satisfaction Component (Normalized Margin)
        # Margin = (Achieved Throughput - GBR) / GBR
        # A value >= 0 means GBR is met or exceeded for that slice.
        # A value < 0 means GBR is violated.
        gbr_margins = (achieved_th - self.slice_gbrs_mbps) / (self.slice_gbrs_mbps + 1e-6) # Add epsilon for GBR=0 case
        
        # The overall throughput safety could be the minimum margin.
        # If any slice violates GBR, this will be negative.
        min_gbr_satisfaction_metric = np.min(gbr_margins)
        # We can clip this to a reasonable range, e.g., [-2, 2]
        # -2 means GBR is violated by 200% or more, +2 means GBR is exceeded by 200% or more.
        min_gbr_satisfaction_metric = np.clip(min_gbr_satisfaction_metric, -2.0, 2.0)

        # 2. Queue Status Component (Normalized "Emptiness")
        # Emptiness = 1 - (Current Queue / Max Queue)
        # A value of 1 means queue is empty (good).
        # A value of 0 means queue is full (bad).
        queue_emptiness = 1.0 - (queue_len / (self.slice_max_queue_kbytes + 1e-6))
        queue_emptiness = np.clip(queue_emptiness, 0.0, 1.0)
        
        # The overall queue safety could be the minimum emptiness.
        min_queue_safety_metric = np.min(queue_emptiness) # Ranges [0, 1]

        # 3. Combine Components into a single safety value m(s')
        # Let's prioritize GBR satisfaction. If GBRs are violated, safety value is primarily negative.
        # If GBRs are met, then queue status can modulate it.
        # A simple approach: If GBRs are met (min_gbr_satisfaction_metric >= 0),
        # safety value starts positive and is further improved by empty queues.
        # If GBRs are violated, safety value is negative and made worse by full queues.

        safety_value = 0.0
        weight_gbr = 2.0  # Weight for GBR satisfaction component
        weight_queue = 1.0 # Weight for queue safety component

        # If GBRs are met/exceeded (metric >=0), treat it as a base positive contribution
        # If GBRs are violated (metric <0), it's a base negative contribution
        safety_value += min_gbr_satisfaction_metric * weight_gbr

        # Add queue component: min_queue_safety_metric ranges [0,1].
        # We want to reward empty queues (metric near 1) and penalize full queues (metric near 0).
        # Let's scale it: (min_queue_safety_metric - 0.5) * 2 -> ranges from -1 (full) to +1 (empty)
        queue_contribution = (min_queue_safety_metric - 0.5) * 2.0
        safety_value += queue_contribution * weight_queue
        
        # Example:
        # GBRs met perfectly (margin=0), queues half full (emptiness=0.5) -> safety = 0*2 + (0.5-0.5)*2*1 = 0
        # GBRs exceeded by 50% (margin=0.5), queues empty (emptiness=1.0) -> safety = 0.5*2 + (1-0.5)*2*1 = 1 + 1 = 2
        # GBRs violated by 50% (margin=-0.5), queues full (emptiness=0.0) -> safety = -0.5*2 + (0-0.5)*2*1 = -1 - 1 = -2

        return float(safety_value)
